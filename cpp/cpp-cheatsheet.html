Reference

Dynamic memory

Prefer smart pointers in modern C++:

auto ptr = std::make_unique<int>(42);

23. Quick Game-Dev Pattern

#include <iostream>

class Player {
public:
    float x = 0;
    float y = 0;
    int health = 100;

    void update() {
        x += 1;
    }

    void draw() {
        std::cout << "Player: "
                  << x << ", "
                  << y << "\n";
    }
};

int main() {
    Player player;

    bool running = true;

    while (running) {
        player.update();
        player.draw();

        if (player.x > 10)
            running = false;
    }

    return 0;
}
